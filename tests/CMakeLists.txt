function(quantum_add_module_contract name library)
    file(GLOB_RECURSE _source "${CMAKE_CURRENT_SOURCE_DIR}/contracts/*/${name}.cpp")
    list(LENGTH _source _count)
    if(NOT _count EQUAL 1)
        message(FATAL_ERROR "Contract ${name} must have exactly one registered source")
    endif()
    add_executable(quantum_${name} ${_source})
    target_include_directories(quantum_${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/support")
    target_link_libraries(quantum_${name} PRIVATE quantum_compile_options ${library})
    quantum_copy_linalg_runtime(quantum_${name})
    add_test(NAME ${name} COMMAND quantum_${name})
endfunction()
